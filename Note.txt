================================================================================
RITUAL WAJIB - PINDAH LAPTOP / FRESH CLONE
================================================================================

cd E:\Github\aplikasi-hub-karyawan\app-karyawan

1. git pull
2. npm install                        # install dependencies
3. docker compose up -d               # jalankan PostgreSQL via Docker (port 5434)
4. npx prisma migrate dev             # sync schema ke DB (development only)
5. npm run prisma:generate            # generate Prisma client
6. npm run prisma:seed:login          # buat akun seed (opsional, jika DB kosong)


================================================================================
DEVELOPMENT (Sehari-hari)
================================================================================

cd E:\Github\aplikasi-hub-karyawan\app-karyawan

# Jalankan frontend + backend sekaligus (mode dev, proxy API via Vite)
npm run dev:full

# Atau pisah:
npm run dev:server    # backend Express di port 4000
npm run dev:host      # frontend Vite di port 5173 (host terbuka LAN)

# Database
npm run db:up         # start Docker PostgreSQL
npm run db:down       # stop Docker PostgreSQL


================================================================================
DEPLOY LOKAL + CLOUDFLARE TUNNEL (Akses Publik)
================================================================================

cd E:\Github\aplikasi-hub-karyawan\app-karyawan

# 1. Build frontend ke dist/ (jalankan tiap ada perubahan kode frontend)
npm run build:prod

##Test

# 2. Jalankan backend (Express melayani dist/ + API di port 4000)
npm run start:prod
# atau dengan PM2:
pm2 start ecosystem.config.cjs

# 3. Jalankan tunnel Cloudflare (mengarah ke port 4000)
cloudflared tunnel --config .cloudflared\cloudflared-hub-karyawan-api.yml run

# Tunnel config:
#   aplikasi-hub.my.id       -> http://127.0.0.1:4000  (static dist/ + admin)
#   www.aplikasi-hub.my.id   -> http://127.0.0.1:4000
#   pwa.aplikasi-hub.my.id   -> http://127.0.0.1:4000
#   api.aplikasi-hub.my.id   -> http://127.0.0.1:4000  (API)

# Catatan: Vite dev server TIDAK perlu jalan saat deploy publik.


================================================================================
PRISMA & DATABASE
================================================================================

npx prisma migrate dev          # buat & jalankan migration baru (dev only)
npx prisma migrate deploy       # jalankan migration di production
npx prisma migrate status       # cek status migration
npx prisma generate             # generate Prisma client
npx prisma studio               # buka Prisma Studio (GUI database)
npm run prisma:generate:safe    # generate Prisma client tanpa matikan backend

npm run security:hash-passwords # konversi password plaintext lama ke hash scrypt
